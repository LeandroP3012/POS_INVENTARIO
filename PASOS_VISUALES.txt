┏━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━┓
┃  PASO 1: INSTALAR INNO SETUP (SOLO UNA VEZ)                          ┃
┗━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━┛

    1. Ir a: https://jrsoftware.org/isdl.php
    
    2. Descargar: innosetup-6.x.x.exe
    
    3. Ejecutar instalador
    
    4. Siguiente > Siguiente > Instalar
    
    ✅ Listo! Inno Setup instalado


┏━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━┓
┃  PASO 2: VERIFICAR QUE TODO ESTÉ LISTO                               ┃
┗━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━┛

    1. Abrir PowerShell en la carpeta del proyecto:
       C:\Users\USER\Desktop\POS
    
    2. Ejecutar:
       
       ┌─────────────────────────────────────┐
       │  python verificar_instalador.py     │
       └─────────────────────────────────────┘
    
    3. Revisar el resultado:
       
       ✅ Todo verde → Continuar al Paso 3
       ❌ Algo rojo → Instalar lo que falte


┏━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━┓
┃  PASO 3: CREAR EL EJECUTABLE (.EXE)                                  ┃
┗━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━┛

    OPCIÓN A - Automático (Recomendado):
    ────────────────────────────────────
    
    Doble clic en:
       ┌─────────────────────────────────────┐
       │  crear_instalador.bat               │
       └─────────────────────────────────────┘
    
    Seguir las instrucciones en pantalla
    Esperar 5-10 minutos
    
    
    OPCIÓN B - Manual:
    ──────────────────
    
    En PowerShell ejecutar:
       ┌─────────────────────────────────────┐
       │  python build_installer.py          │
       └─────────────────────────────────────┘
    
    Esperar a que finalice (5-10 minutos)
    
    
    📁 Resultado:
       dist\POS_Sistema\POS_Sistema.exe
       
    ⚠️  Puedes probar este .exe, pero aún no es el instalador


┏━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━┓
┃  PASO 4: CREAR EL INSTALADOR (.EXE INSTALADOR)                       ┃
┗━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━┛

    1. Abrir Inno Setup Compiler
       (Menú Inicio > Inno Setup > Inno Setup Compiler)
    
    2. File > Open
    
    3. Seleccionar:
       C:\Users\USER\Desktop\POS\installer_script.iss
    
    4. Build > Compile (o presionar F9)
       
       ┌─────────────────────────────────────────┐
       │  🔨 Compilando instalador...            │
       │                                         │
       │  [████████████████████        ] 85%     │
       │                                         │
       │  Por favor espere...                    │
       └─────────────────────────────────────────┘
    
    5. Esperar 1-2 minutos
    
    6. ¡Listo!
       
       ┌─────────────────────────────────────────┐
       │  ✅ Compile complete                    │
       │                                         │
       │  Output file:                           │
       │  Output\POS_Setup.exe                   │
       └─────────────────────────────────────────┘

    📦 C:\Users\USER\Desktop\POS\Output\POS_Setup.exe

    ─────────────────────────

       → Compartir link de descarga

       → Solo si es <25 MB (comprimido)
       → Usar WeTransfer para archivos grandes
    
    💿 CD/DVD
       → Grabar con ImgBurn o similar

    REQUISITOS DE LA PC DESTINO:
    ─────────────────────────────
    ✅ Windows 10 o 11 (64 bits)
    ✅ MySQL Server 8.0+ (instalado)
    ✅ 4 GB RAM mínimo
    ✅ 500 MB espacio disco
    
    
    PASOS PARA EL USUARIO FINAL:
    ────────────────────────────
    
    1️⃣  Instalar MySQL Server (si no lo tiene)
        https://dev.mysql.com/downloads/mysql/
    
    2️⃣  Crear base de datos:
        mysql> CREATE DATABASE pos_db;
    
    3️⃣  Ejecutar POS_Setup.exe
        - Doble clic
        - Siguiente > Siguiente
        - Ingresar datos de MySQL:
          • Host: localhost
          • Puerto: 3306
          • Usuario: root
          • Contraseña: (su contraseña)
          • Base datos: pos_db
    
    4️⃣  Ejecutar scripts SQL:
        Ir a: C:\Program Files\Sistema POS\database\
        Ejecutar en orden:
        - create_tables.sql
        - add_roles_system.sql
        - add_user_sessions_table.sql
    
    5️⃣  Abrir Sistema POS
        Escritorio > Sistema POS
        Usuario: admin
        Password: admin123
